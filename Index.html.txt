    function handleReferral() {
      if (!tg?.initDataUnsafe?.start_param) return;
      const param = tg.initDataUnsafe.start_param;
      if (!param.startsWith('ref_')) return;

      const refId = param.replace('ref_', '');

      } catch(e) {}

      save();
    }

    function claimPendingReferrals() {

    function formatTime(sec) {

        state.level++;
        state.maxEnergy = 100 + (state.level - 1) * 10;
        state.energy = state.maxEnergy;
        leveled = true;

      if (leveled) {
        showToast(`🎉 Уровень ${state.level}!`);
        haptic('success');
      }
      save();
      render();
    }

      return Math.min(1, elapsed / effectiveTime);

      const strain = getStrain(pot.strain);
      const effectiveTime = strain.growTime / (pot.speedMult || 1);
      const elapsed = (Date.now() - pot.plantedAt) / 1000;
      return Math.max(0, Math.ceil(effectiveTime - elapsed));
    }

    // ==================== WATERING ====================
    function waterPot(potId) {
      const pot = state.pots.find(p => p.id === potId);
      if (!pot || !pot.strain || pot.watered) return;

      if (state.energy < 3) {
        showToast('Нужно 3⚡ для полива', true);
        haptic('error');
        return;
      }

      state.energy -= 3;
      pot.watered = true;
      // Watering gives small speed boost
      pot.speedMult = (pot.speedMult || 1) * 1.15;

      showToast('💧 Полили! Рост ускорен');
      haptic('medium');
      addXp(2);
      save();
      render();
    }

    // ==================== FERTILIZERS ====================
    function openFertModal(potId) {
      selectedPotId = potId;
      const list = document.getElementById('fert-list');
      list.innerHTML = '';

      // Owned fertilizers
      let hasAny = false;
      FERTILIZERS.forEach(f => {
        const count = state.fertilizers[f.id] || 0;
        if (count <= 0) return;
        hasAny = true;

        const card = document.createElement('div');
        card.className = 'card';
        card.innerHTML = `
          <div class="card-emoji">${f.emoji}</div>
          <div class="card-info">

            <div class="card-desc">${f.desc}</div>
          </div>
          <button class="btn btn-small btn-warning" onclick="applyFert('${f.id}')">Применить</button>
        `;
        list.appendChild(card);
      });

      if (!hasAny) {
        list.innerHTML = `<div class="empty"><div class="empty-emoji">🧪</div><div>Нет удобрений.<br>Купи их в магазине!</div></div>`;
      }

      document.getElementById('modal-fert').classList.add('show');
    }

    function applyFert(fertId) {
      const pot = state.pots.find(p => p.id === selectedPotId);
      if (!pot || !pot.strain) return;
      if ((state.fertilizers[fertId] || 0) <= 0) return;

      const fert = getFert(fertId);
      state.fertilizers[fertId]--;

      if (fert.effect === 'speed' || fert.effect === 'both') {
        pot.speedMult = (pot.speedMult || 1) * (1 + fert.value);
      }
      if (fert.effect === 'yield' || fert.effect === 'both') {
        pot.yieldMult = (pot.yieldMult || 1) * (1 + fert.value);
      }
      pot.fert = fertId;

      closeModal('modal-fert');
      showToast(`${fert.emoji} ${fert.name} применено!`);
      haptic('success');
      addXp(5);
      save();
      render();
    }

    function buyFertilizer(fertId) {
      const fert = getFert(fertId);
      if (state.money < fert.price) {
        showToast('Не хватает денег!', true);
        haptic('error');
        return;
      }
      state.money -= fert.price;
      state.fertilizers[fertId] = (state.fertilizers[fertId] || 0) + 1;
      showToast(`Куплено: ${fert.name}`);
      haptic('medium');
      save();
      render();
    }

    // ==================== PLANTING ====================
    function openSeedModal(potId) {
      selectedPotId = potId;
      const list = document.getElementById('seed-list');
      list.innerHTML = '';

      STRAINS.forEach(s => {
        const unlocked = state.level >= s.unlockLevel;
        const canBuy = state.money >= s.seedPrice && state.energy >= s.energyCost;

        const card = document.createElement('div');
        card.className = 'card';
        card.style.opacity = unlocked ? '1' : '0.45';
        card.innerHTML = `

          </div>
          <div style="text-align:right">
            <div class="card-price">${s.seedPrice}$</div>
            <div style="font-size:11px;color:var(--hint)">⚡${s.energyCost}</div>
          </div>
        `;
        if (unlocked && canBuy) {
          card.style.cursor = 'pointer';
          card.onclick = () => plantSeed(s.id);
        }
        list.appendChild(card);
      });

      document.getElementById('modal-seed').classList.add('show');
    }

    function closeModal(id) {
      document.getElementById(id).classList.remove('show');
    }

    function plantSeed(strainId) {
      const strain = getStrain(strainId);
      if (!strain) return;
      if (state.money < strain.seedPrice || state.energy < strain.energyCost) {
        showToast('Не хватает ресурсов!', true);
        haptic('error');
        return;
      }

      const pot = state.pots.find(p => p.id === selectedPotId);
      if (!pot || pot.strain) return;

      state.money -= strain.seedPrice;
      state.energy -= strain.energyCost;
      pot.strain = strainId;
      pot.plantedAt = Date.now();
      pot.watered = false;
      pot.fert = null;
      pot.speedMult = 1;
      pot.yieldMult = 1;

      closeModal('modal-seed');
      showToast(`Посажено: ${strain.name}`);
      haptic('medium');
      addXp(5);
      save();
      render();
    }

    function harvest(potId) {
      const pot = state.pots.find(p => p.id === potId);
      if (!pot || !pot.strain) return;

      const progress = getGrowProgress(pot);
      if (progress < 1) {
        showToast('Ещё не выросло!', true);
        return;
      }

      const strain = getStrain(pot.strain);
      let amount = Math.floor(strain.yield * (pot.yieldMult || 1));
      // Bonus if watered
      if (pot.watered) amount = Math.floor(amount * 1.1);

      state.inventory[pot.strain] = (state.inventory[pot.strain] || 0) + amount;

      showToast(`Собрано ${amount} × ${strain.name}!`);
      haptic('success');
      addXp(amount * 3);

      pot.strain = null;
      pot.plantedAt = null;
      pot.watered = false;
      pot.fert = null;
      pot.speedMult = 1;
      pot.yieldMult = 1;

      save();
      render();
    }

    // ==================== SELLING ====================
    function sellItem(strainId) {
      const amount = state.inventory[strainId] || 0;
      if (amount <= 0) return;
      const strain = getStrain(strainId);
      const total = amount * strain.sellPrice;
      state.money += total;
      delete state.inventory[strainId];
      showToast(`Продано за ${total}$`);
      haptic('success');
      addXp(Math.floor(total / 10));
      save();
      render();
    }

    function sellAll() {
      let total = 0, items = 0;
      for (const [id, amount] of Object.entries(state.inventory)) {
        const strain = getStrain(id);
        if (strain && amount > 0) {
          total += amount * strain.sellPrice;
          items += amount;
        }
      }
      if (items === 0) {
        showToast('Склад пуст!', true);
        return;
      }
      state.money += total;
      state.inventory = {};
      showToast(`Продано всё за ${total}$`);
      haptic('success');
      addXp(Math.floor(total / 10));
      save();
      render();
    }

    // ==================== CUSTOMERS ====================
    function spawnCustomer() {
      if (state.currentCustomer) return;
      if (Date.now() - state.lastCustomerAt < 45000) return; // min 45 sec between

      // Need something to sell
      const hasStock = Object.values(state.inventory).some(v => v > 0);
      if (!hasStock) return;

      const cust = CUSTOMERS[Math.floor(Math.random() * CUSTOMERS.length)];
      // Pick what they want
      let wantId = cust.prefer;
      if (!wantId || !(state.inventory[wantId] > 0)) {
        // pick random from inventory
        const available = Object.keys(state.inventory).filter(k => state.inventory[k] > 0);
        if (available.length === 0) return;
        wantId = available[Math.floor(Math.random() * available.length)];
      }

      const amount = Math.min(state.inventory[wantId], 1 + Math.floor(Math.random() * 3));
      const strain = getStrain(wantId);
      const price = Math.floor(strain.sellPrice * amount * cust.mult);

      state.currentCustomer = {
        ...cust,
        wantId,
        amount,
        price,
        expiresAt: Date.now() + 35000
      };
      state.lastCustomerAt = Date.now();

      showCustomerBanner();
      haptic('warning');
    }

    function showCustomerBanner() {
      const c = state.currentCustomer;
      if (!c) return;
      const strain = getStrain(c.wantId);
      document.getElementById('cust-avatar').textContent = c.emoji;
      document.getElementById('cust-name').textContent = c.name;
      document.getElementById('cust-want').textContent =
        `Хочет ${c.amount}× ${strain.name} за ${c.price}$ (${Math.round((c.mult - 1) * 100)}% бонус)`;
      document.getElementById('customer-banner').classList.add('show');
    }

    function hideCustomerBanner() {
      document.getElementById('customer-banner').classList.remove('show');
      state.currentCustomer = null;
      save();
    }

    function dealWithCustomer() {
      const c = state.currentCustomer;
      if (!c) return;

      if ((state.inventory[c.wantId] || 0) < c.amount) {
        showToast('Недостаточно товара!', true);
        haptic('error');
        return;
      }

      state.inventory[c.wantId] -= c.amount;
      if (state.inventory[c.wantId] <= 0) delete state.inventory[c.wantId];
      state.money += c.price;

      showToast(`Сделка с ${c.name}! +${c.price}$`);
      haptic('success');
      addXp(Math.floor(c.price / 8));
      hideCustomerBanner();
      render();
    }

    function smokeWithCustomer() {
      const c = state.currentCustomer;
      if (!c) return;

      // Smoking restores energy and slightly improves deal
      state.energy = Math.min(state.maxEnergy, state.energy + c.smokeBonus);
      c.mult += 0.15;
      c.price = Math.floor(getStrain(c.wantId).sellPrice * c.amount * c.mult);

      document.getElementById('cust-want').textContent =
        `Хочет ${c.amount}× ${getStrain(c.wantId).name} за ${c.price}$ (после покура)`;

      showToast(`🚬 Покурили! +${c.smokeBonus}⚡ и лучшая цена`);
      haptic('medium');
      save();
      renderHeader();
    }

    function rejectCustomer() {
      showToast(`${state.currentCustomer?.name || 'Клиент'} ушёл...`);
      hideCustomerBanner();
      haptic('light');
    }

    // ==================== POTS & ENERGY ====================
    function buyPot() {
      const price = 300 + state.pots.length * 200;
      if (state.money < price) {
        showToast('Не хватает денег!', true);
        haptic('error');
        return;
      }
      state.money -= price;
      state.pots.push({
        id: Date.now(),
        strain: null,
        plantedAt: null,
        watered: false,
        fert: null,
        speedMult: 1,
        yieldMult: 1
      });
      showToast('Новый горшок!');
      haptic('success');
      addXp(10);
      save();
      render();
    }

    function buyEnergy() {
      if (state.money < 100) {
        showToast('Не хватает денег!', true);
        return;
      }
      state.money -= 100;
      state.energy = Math.min(state.maxEnergy, state.energy + 40);
      showToast('+40 энергии');
      haptic('medium');
      save();
      render();
    }

    // ==================== RENDER ====================
    function renderFarm() {
      const grid = document.getElementById('farm-grid');
      grid.innerHTML = '';

      state.pots.forEach(pot => {
        const div = document.createElement('div');
        div.className = 'pot';

        if (!pot.strain) {
          div.innerHTML = `
            <div class="pot-emoji">🪴</div>
            <div class="pot-name">Пустой горшок</div>
            <div class="pot-status">Готов к посадке</div>
            <button class="btn btn-full" onclick="openSeedModal(${pot.id})">Посадить</button>
          `;
        } else {
          const strain = getStrain(pot.strain);
          const progress = getGrowProgress(pot);
          const remaining = getRemaining(pot);
          const stage = Math.min(4, Math.floor(progress * 4));
          const isReady = progress >= 1;

          if (isReady) div.classList.add('ready');
          else if (!pot.watered) div.classList.add('needs-water');

          let boosts = '';
          if (pot.watered) boosts += '<span class="boost-tag">💧</span>';
          if (pot.fert) boosts += '<span class="boost-tag fert">🧪</span>';

          let actions = '';
          if (isReady) {
            actions = `<button class="btn btn-full" onclick="harvest(${pot.id})">🌿 Собрать</button>`;
          } else {
            actions = `<div class="pot-actions">`;
            if (!pot.watered) {
              actions += `<button class="btn btn-blue btn-small" onclick="waterPot(${pot.id})">💧 Полить</button>`;
            }
            actions += `<button class="btn btn-warning btn-small" onclick="openFertModal(${pot.id})">🧪</button>`;
            actions += `</div>`;
          }

          div.innerHTML = `
            <div class="pot-emoji">${isReady ? '✨' : STAGE_EMOJIS[stage]}</div>
            <div class="pot-name">${strain.name}${boosts}</div>
            <div class="pot-status">${isReady ? 'Готово!' : formatTime(remaining)}</div>
            <div class="progress-bar">
              <div class="progress-fill" style="width:${progress * 100}%"></div>
            </div>
            ${actions}
          `;
        }
        grid.appendChild(div);
      });

      const price = 300 + state.pots.length * 200;
      document.getElementById('btn-buy-pot').textContent = `➕ Купить горшок (${price}$)`;
    }
